# DriftFinder Experiment — Complete Step by Step Protocol
# Every single step is written out. No shortcuts. No "repeat steps X-Y".
# Follow this from top to bottom without skipping anything.

=======================================================================
SECTION 1: SETUP (DO ONCE)
=======================================================================

STEP 1
Open your terminal and run:
aws sts get-caller-identity
Copy your Account ID from the output. You will need it throughout.

STEP 2
In AWS Console go to Billing > Budgets and set a $50 alert on your account.

STEP 3
In your project root, create a results folder:
mkdir -p results

STEP 4
Copy the three config files below into your project root.

File: .driftfinder-terraform.yml
---
iac_tool: terraform
state_file: experiment/terraform/terraform.tfstate
aws_region: eu-west-1
fail_on: LOW
output_format: json
---

File: .driftfinder-cloudformation.yml
---
iac_tool: cloudformation
stack_name: driftfinder-experiment-cf
aws_region: eu-west-1
fail_on: LOW
output_format: json
---

File: .driftfinder-pulumi.yml
---
iac_tool: pulumi
pulumi_stack: YOUR_ORG/driftfinder-experiment/driftfinder-experiment
aws_region: eu-west-1
fail_on: LOW
output_format: json
---

=======================================================================
SECTION 2: PROVISION THREE ENVIRONMENTS (DO ONCE)
=======================================================================

--- TERRAFORM ---

STEP 5
cd experiment/terraform

STEP 6
terraform init

STEP 7
terraform apply \
  -var="account_id=YOUR_ACCOUNT_ID" \
  -var="db_password=DriftFinder123!" \
  -var="region=eu-west-1"
Type yes when prompted. Wait 10-15 minutes for RDS to finish.

STEP 8
terraform output -json > ../resources_terraform.json

STEP 9
cd ../..

--- CLOUDFORMATION ---

STEP 10
aws cloudformation deploy \
  --template-file experiment/cloudformation/baseline.yaml \
  --stack-name driftfinder-experiment-cf \
  --parameter-overrides AccountId=YOUR_ACCOUNT_ID DBPassword=DriftFinder123! \
  --capabilities CAPABILITY_NAMED_IAM \
  --region eu-west-1
Wait 10-15 minutes for RDS to finish.

STEP 11
aws cloudformation describe-stacks \
  --stack-name driftfinder-experiment-cf \
  --region eu-west-1 \
  --query "Stacks[0].Outputs" \
  --output json > experiment/resources_cloudformation_raw.json

STEP 12
python3 -c "
import json
raw = json.load(open('experiment/resources_cloudformation_raw.json'))
key_map = {
    'S3BucketName': 's3_bucket_name',
    'SecurityGroupId': 'security_group_id',
    'IAMPolicyArn': 'iam_policy_arn',
    'RDSInstanceId': 'rds_instance_id',
    'EBSVolumeId': 'ebs_volume_id',
    'CloudTrailName': 'cloudtrail_name',
    'VPCId': 'vpc_id',
    'KMSKeyId': 'kms_key_id',
    'KMSKeyArn': 'kms_key_arn',
}
out = {key_map.get(o['OutputKey'], o['OutputKey']): o['OutputValue'] for o in raw}
json.dump(out, open('experiment/resources_cloudformation.json', 'w'), indent=2)
print('Done')
"

--- PULUMI ---

STEP 13
cd experiment/pulumi

STEP 14
pip install pulumi pulumi-aws

STEP 15
pulumi stack init driftfinder-experiment

STEP 16
pulumi config set aws:region eu-west-1

STEP 17
pulumi config set account_id YOUR_ACCOUNT_ID

STEP 18
pulumi config set --secret db_password DriftFinder123!

STEP 19
pulumi up
Type yes when prompted. Wait 10-15 minutes for RDS.

STEP 20
pulumi stack output --json > ../resources_pulumi.json

STEP 21
cd ../..
